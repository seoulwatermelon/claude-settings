     1→     1→# TO 배정 대시보드 Implementation Plan
     2→     2→
     3→     3→> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.
     4→     4→
     5→     5→**Goal:** 메인 페이지(`/`)에 학과별 TO 배정 현황을 엑셀 표 형식으로 보여주고, 관리자가 Excel 업로드 또는 인라인 편집으로 데이터를 관리할 수 있게 한다.
     6→     6→
     7→     7→**Architecture:** `to_haejung` SQLite 테이블에 학과별 데이터를 저장하고, Next.js App Router API 라우트 3개(GET/POST upload/PATCH)로 CRUD를 처리한다. `TOHaejungTable` 컴포넌트가 읽기 전용(메인)/편집(관리자) 두 모드를 하나의 prop으로 처리한다.
     8→     8→
     9→     9→**Tech Stack:** Next.js 13 App Router, TypeScript, Tailwind CSS, better-sqlite3, xlsx@0.18.5

   443→   443→    formData = await req.formData();
   444→   444→  } catch {
   445→   445→    return NextResponse.json({ error: '요청 파싱 실패' }, { status: 400 });
   446→   446→  }
   447→   447→

   449→   449→  if (!file) {
   450→   450→    return NextResponse.json({ error: '파일이 없습니다' }, { status: 400 });
   451→   451→  }
   452→   452→
   453→   453→  // 파일 크기 10MB 제한
   454→   454→  if (file.size > 10 * 1024 * 1024) {
   455→   455→    return NextResponse.json({ error: '파일 크기가 10MB를 초과합니다' }, { status: 400 });
   456→   456→  }
   457→   457→
   458→   458→  // 확장자 및 MIME 검증
   459→   459→  const isXlsx =
   460→   460→    file.name.toLowerCase().endsWith('.xlsx') ||
   461→   461→    file.type === 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet';
   462→   462→  if (!isXlsx) {
   463→   463→    return NextResponse.json({ error: '.xlsx 파일만 업로드 가능합니다' }, { status: 400 });
   464→   464→  }
   465→   465→

   467→   467→    const buffer = Buffer.from(await file.arrayBuffer());
   468→   468→    const wb = XLSX.read(buffer, { type: 'buffer' });
   469→   469→    const ws = wb.Sheets[wb.SheetNames[0]];
   470→   470→    const rawRows = XLSX.utils.sheet_to_json<Record<string, unknown>>(ws, { defval: null });
   471→   471→
   472→   472→    // 필수 컬럼 검증
   473→   473→    if (rawRows.length > 0) {
   474→   474→      const headers = Object.keys(rawRows[0]);
   475→   475→      if (!headers.includes('소속대학') || !headers.includes('구분')) {
   476→   476→        return NextResponse.json(
   477→   477→          { error: '필수 컬럼(소속대학, 구분)이 없습니다. RAW 데이터 형식의 파일을 업로드하세요.' },
   478→   478→          { status: 400 }
   479→   479→        );
   480→   480→      }
   481→   481→    }

   486→   486→    for (const raw of rawRows) {

   497→   497→        seen.add(key);

   499→   499→      }

   505→   505→    const insertStmt = db.prepare(
   506→   506→      `INSERT INTO to_haejung (${insertCols}) VALUES (${placeholders})`
   507→   507→    );

   510→   510→      db.prepare('DELETE FROM to_haejung').run();

   513→   513→        insertStmt.run(values);
   514→   514→      }

   521→   521→      { error: err instanceof Error ? err.message : '처리 실패' },
   522→   522→      { status: 500 }

   524→   524→  }
