     1→# TO 배정 대시보드 Implementation Plan
     2→
     3→> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.
     4→
     5→**Goal:** 메인 페이지(`/`)에 학과별 TO 배정 현황을 엑셀 표 형식으로 보여주고, 관리자가 Excel 업로드 또는 인라인 편집으로 데이터를 관리할 수 있게 한다.
     6→
     7→**Architecture:** `to_haejung` SQLite 테이블에 학과별 데이터를 저장하고, Next.js App Router API 라우트 3개(GET/POST upload/PATCH)로 CRUD를 처리한다. `TOHaejungTable` 컴포넌트가 읽기 전용(메인)/편집(관리자) 두 모드를 하나의 prop으로 처리한다.
     8→
     9→**Tech Stack:** Next.js **14** App Router, TypeScript, Tailwind CSS, better-sqlite3, xlsx@0.18.5
    10→
    11→> **Next.js 버전 참고:** 설치된 버전은 `^14.0.0`. `params`는 Next.js 15부터 Promise이므로, `{ params: { id: string } }` 동기 타입이 올바름.

   494→    return NextResponse.json({ error: '파일이 없습니다' }, { status: 400 });
   495→  }
   496→
   497→  // 파일 크기 10MB 제한
   498→  if (file.size > 10 * 1024 * 1024) {
   499→    return NextResponse.json({ error: '파일 크기가 10MB를 초과합니다' }, { status: 400 });
   500→  }
   501→
   502→  // 확장자 AND MIME 검증 (둘 다 만족해야 통과 — 우회 방지)
   503→  const isXlsx =
   504→    file.name.toLowerCase().endsWith('.xlsx') &&
   505→    file.type === 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet';
   506→  if (!isXlsx) {
   507→    return NextResponse.json({ error: '.xlsx 파일만 업로드 가능합니다' }, { status: 400 });
   508→  }
   509→
   510→  try {
   511→    const buffer = Buffer.from(await file.arrayBuffer());
   512→    const wb = XLSX.read(buffer, { type: 'buffer' });
   513→    const ws = wb.Sheets[wb.SheetNames[0]];
   514→    const rawRows = XLSX.utils.sheet_to_json<Record<string, unknown>>(ws, { defval: null });
   515→
   516→    // 필수 컬럼 검증
   517→    if (rawRows.length > 0) {
   518→      const headers = Object.keys(rawRows[0]);
   519→      if (!headers.includes('소속대학') || !headers.includes('구분')) {
   520→        return NextResponse.json(
   521→          { error: '필수 컬럼(소속대학, 구분)이 없습니다. RAW 데이터 형식의 파일을 업로드하세요.' },
   522→          { status: 400 }
   523→        );
   524→      }
   525→    }

   527→    // 매핑 + 중복 제거 (college, department 기준)

   529→    const rows: Record<string, string | null>[] = [];
   530→    for (const raw of rawRows) {
   531→      const row: Record<string, string | null> = {};
   532→      for (const [excelHeader, dbCol] of Object.entries(COLUMN_MAP)) {
   533→        row[dbCol] = toText(raw[excelHeader]);
   534→      }
   535→      // department fallback
   536→      if (!row['department']) {
   537→        row['department'] = toText(raw['학과명']);
   538→      }
   539→      const key = `${row['college']}||${row['department']}`;
   540→      if (!seen.has(key)) {
   541→        seen.add(key);
   542→        rows.push(row);
   543→      }
   544→    }

   547→    const insertCols = DB_COLUMNS.join(', ');
   548→    const placeholders = DB_COLUMNS.map(() => '?').join(', ');
   549→    const insertStmt = db.prepare(

   551→    );
   552→
   553→    const replaceAll = db.transaction(() => {
   554→      db.prepare('DELETE FROM to_haejung').run();
   555→      for (const row of rows) {
   556→        const values = DB_COLUMNS.map((col) => row[col] ?? null);
   557→        insertStmt.run(values);
   558→      }
   559→    });
   560→    replaceAll();
   561→
   562→    return NextResponse.json({ ok: true, count: rows.length });
   563→  } catch (err) {
   564→    return NextResponse.json(
   565→      { error: err instanceof Error ? err.message : '처리 실패' },
   566→      { status: 500 }
   567→    );
   568→  }
   569→}

   732→    }
   733→  }
