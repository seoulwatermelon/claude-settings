    52→function formatValue(value: string | number | null | undefined) {
    53→  if (value === null || value === undefined || value === "") return "-";
    54→  if (typeof value === "number") return Number.isInteger(value) ? String(value) : value.toFixed(2);
    55→  return value;
    56→}
    57→
    58→function formatCellValue(columnKey: string, value: string | number | null | undefined) {
    59→  return formatValue(value);

    62→function getCurrentRequest(detail: DashboardDetail | null, selectedRequestId: number | null) {
    63→  if (!detail || !detail.recruitmentRequests.length) return null;
    64→  return (
    65→    detail.recruitmentRequests.find((request) => request.id === selectedRequestId) ??
    66→    detail.recruitmentRequests[0]
    67→  );
    68→}

    70→function getRequestProgressMeta(request: RecruitmentRequest) {
    71→  const procedureType = getProcedureType(request.hiringType);
    72→  const steps = PROCEDURE_STEPS[procedureType];
    73→  const currentIndex = Math.max(steps.findIndex((step) => step.stepKey === request.status), 0);
    74→  return {
    75→    stepTitle: steps[currentIndex]?.title ?? request.status,
    76→    percent: Math.round(((currentIndex + 1) / steps.length) * 100),
    77→  };
    78→}
