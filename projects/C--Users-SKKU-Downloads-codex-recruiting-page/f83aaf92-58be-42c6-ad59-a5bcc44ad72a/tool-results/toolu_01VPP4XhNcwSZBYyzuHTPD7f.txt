    97→function compactValues(values: Array<string | number | null | undefined>) {
    98→  const seen = new Set<string>();
    99→  const result: string[] = [];
   100→
   101→  for (const value of values) {
   102→    if (value === null || value === undefined || value === "") {
   103→      continue;
   104→    }
   105→
   106→    const text = String(value).trim();
   107→    if (!text || text === "None" || text === "nan") {
   108→      continue;
   109→    }
   110→
   111→    if (!seen.has(text)) {
   112→      seen.add(text);
   113→      result.push(text);
   114→    }
   115→  }
   116→
   117→  return result;
   118→}

   163→function aggregateDepartmentRows(rows: DashboardRow[]): DashboardRow {
   164→  const [firstRow] = rows;
   165→  const allKeys = new Set<string>();
   166→  rows.forEach((row) => Object.keys(row.values).forEach((key) => allKeys.add(key)));
   167→
   168→  const aggregatedValues: Record<string, string | number | null> = {};
   169→  for (const key of allKeys) {
   170→    const compacted = compactValues(rows.map((row) => row.values[key]));
   171→    aggregatedValues[key] = compacted.length <= 1 ? (compacted[0] ?? null) : compacted.join(" / ");
   172→  }
   173→
   174→  const metricRows = rows.reduce<Record<string, DashboardRow>>((acc, row) => {
   175→    const label = row.values[METRIC_LABEL_COLUMN];
   176→    if (typeof label === "string" && label.trim()) {
   177→      acc[label.trim()] = row;
   178→    }
   179→    return acc;
   180→  }, {});
   181→
   182→  const countRow = metricRows["교원수"];
   183→  const rateRow = metricRows["확보율"];
   184→  const metricTypeRow = metricRows["구분"];
   185→  const metricValueRow = metricRows["학점"];
   186→
   187→  aggregatedValues["교원수 24년"] = countRow?.values[METRIC_YEAR_COLUMNS[0]] ?? null;
   188→  aggregatedValues["교원수 25년"] = countRow?.values[METRIC_YEAR_COLUMNS[1]] ?? null;
   189→  aggregatedValues["교원수 26년"] = countRow?.values[METRIC_YEAR_COLUMNS[2]] ?? null;
   190→  aggregatedValues["교원수 증감"] = countRow?.values[METRIC_YEAR_COLUMNS[3]] ?? null;
   191→  aggregatedValues["확보율 24년"] = rateRow?.values[METRIC_YEAR_COLUMNS[0]] ?? null;
   192→  aggregatedValues["확보율 25년"] = rateRow?.values[METRIC_YEAR_COLUMNS[1]] ?? null;
   193→  aggregatedValues["확보율 26년"] = rateRow?.values[METRIC_YEAR_COLUMNS[2]] ?? null;
   194→  aggregatedValues["확보율 증감"] = rateRow?.values[METRIC_YEAR_COLUMNS[3]] ?? null;
   195→
   196→  const metricNames = [
   197→    metricTypeRow?.values[METRIC_YEAR_COLUMNS[0]],
   198→    metricTypeRow?.values[METRIC_YEAR_COLUMNS[1]],
   199→    metricTypeRow?.values[METRIC_YEAR_COLUMNS[2]],
   200→  ];
   201→  const metricValues = [
   202→    metricValueRow?.values[METRIC_YEAR_COLUMNS[0]],
   203→    metricValueRow?.values[METRIC_YEAR_COLUMNS[1]],
   204→    metricValueRow?.values[METRIC_YEAR_COLUMNS[2]],
   205→  ];
   206→
   207→  metricNames.forEach((name, index) => {
   208→    const normalized = typeof name === "string" ? name.trim() : "";
   209→    if (normalized === "책임학점") aggregatedValues["책임학점"] = metricValues[index] ?? null;
   210→    if (normalized === "강의학점") aggregatedValues["강의학점"] = metricValues[index] ?? null;
   211→    if (normalized === "초과율") aggregatedValues["초과율"] = metricValues[index] ?? null;
   212→  });
   213→
   214→  return {
   215→    id: firstRow.id,
   216→    deptCode: firstRow.deptCode,
   217→    deptName: firstRow.deptName,
   218→    college: firstRow.college,
   219→    rowOrder: firstRow.rowOrder,
   220→    requestId: null,
   221→    values: aggregatedValues,
   222→  };
   223→}

   235→function getProgressSortScore(request: RecruitmentRequest | null) {
   236→  if (!request) return -1;
   237→  const steps = PROCEDURE_STEPS[getProcedureType(request.hiringType)];
   238→  const stepIndex = getProgressStageIndex(request);
   239→  return ((stepIndex + 1) / Math.max(steps.length, 1)) * 100;

   316→function getDepartmentBaseRows() {
   317→  const db = getDb();
   318→  const rows = db
   319→    .prepare(
   320→      `SELECT id, dept_code as deptCode, dept_name as deptName, college, row_order as rowOrder, raw_json as rawJson
   321→       FROM dashboard_rows
   322→       ORDER BY row_order ASC`,
   323→    )
   324→    .all() as Array<Omit<DashboardRow, "values" | "requestId"> & { rawJson: string }>;
   325→
   326→  const parsed = rows.map(({ rawJson, ...row }) => ({
   327→    ...row,
   328→    requestId: null,
   329→    values: parseRowJson(rawJson),
   330→  }));
   331→  const grouped = new Map<string, DashboardRow[]>();
   332→  for (const row of parsed) {
   333→    const key = row.deptCode ?? `${row.college ?? ""}::${row.deptName ?? row.id}`;
   334→    const bucket = grouped.get(key) ?? [];
   335→    bucket.push(row);
   336→    grouped.set(key, bucket);
   337→  }
   338→
   339→  return [...grouped.values()].map(aggregateDepartmentRows);
   340→}

   356→      deptCode: deptRow.deptCode,
   357→      deptName: deptRow.deptName,
   358→      college: deptRow.college,
   359→      rowOrder: deptRow.rowOrder + Number(request.priority ?? 0) / 100,
   360→      requestId: request.id,
   361→      values: {
   362→        ...deptRow.values,
   363→        "요   청 일반": request.requestedGeneral ?? null,
   364→        "요   청 전담": request.requestedDedicated ?? null,
   365→        "요   청 外": request.requestedForeign ?? null,
   366→        "요   청 계":
   367→          request.requestedCount ??
   368→          (request.requestedGeneral ?? 0) +
   369→            (request.requestedDedicated ?? 0) +
   370→            (request.requestedForeign ?? 0),
   371→        "(최초신청) 초빙분야": request.fieldName,
   372→        "(최초신청) 학과요청사유": request.deptRequestReason ?? request.hiringReason,
   373→        "(최종) 초빙분야": request.fieldName,
   374→        채용구분: request.hiringType,
   375→        현재단계: getProgressStageLabel(request),
   376→        "학과장 액션": "조치 필요",
   377→        진행률: `${Math.round(((getProgressStageIndex(request) + 1) / PROCEDURE_STEPS[getProcedureType(request.hiringType)].length) * 100)}%`,
   378→        ...stageCells,
   379→      },
   380→    });
   381→  }
   382→
   383→  const deptsWithRequests = new Set(rows.map((row) => row.deptCode));
   384→  for (const deptRow of deptRows) {
   385→    if (deptsWithRequests.has(deptRow.deptCode)) continue;
   386→    rows.push({
   387→      ...deptRow,
   388→      id: deptRow.id,
   389→      rowOrder: fallbackRowOrder,
   390→      values: {
   391→        ...deptRow.values,
   392→        채용구분: null,
   393→        현재단계: null,
   394→        "학과장 액션": null,
   395→        진행률: null,
   396→        기획: null,
   397→        요청: null,
   398→        승인: null,
   399→        세부심사: null,
   400→        협의: null,
   401→        임용준비: null,
   402→        완료: null,
   403→      },
   404→    });
   405→    fallbackRowOrder += 1;
   406→  }
   407→
   408→  return rows.sort((a, b) => {
   409→    const requestA = a.requestId ? requests.find((request) => request.id === a.requestId) ?? null : null;
   410→    const requestB = b.requestId ? requests.find((request) => request.id === b.requestId) ?? null : null;
   411→    const progressDiff = getProgressSortScore(requestB) - getProgressSortScore(requestA);
   412→    if (progressDiff !== 0) return progressDiff;
   413→    return a.rowOrder - b.rowOrder;
   414→  });
   415→}

   431→  const db = getDb();

   434→      `SELECT column_key as key, group_label as groupLabel, column_label as columnLabel, display_order as displayOrder

   461→    }
   462→
   463→    if (HIDE_COLUMNS.has(sourceColumn.key)) continue;

   466→      ...sourceColumn,

   469→      displayOrder: transformed.length + 1,

   740→function getStepPayloads(request: RecruitmentRequest, submissions: SubmissionRecord[]) {
   741→  const stepPayloads: Record<string, WorkflowStepPayload> = {};
   742→  const stepForms: Record<string, ReturnType<typeof getWorkflowConfigForRequest>> = {};
   743→  const procedureType = getProcedureType(request.hiringType);
   744→  const workflow = WORKFLOW_FORMS[procedureType];
   745→
   746→  for (const step of PROCEDURE_STEPS[procedureType]) {
   747→    stepForms[step.stepKey] = workflow[step.stepKey] ?? null;
   748→  }
   749→
   750→  for (const submission of submissions) {
   751→    if (stepPayloads[submission.stepKey]) continue;
   752→    stepPayloads[submission.stepKey] = {
   753→      stepKey: submission.stepKey,
   754→      values: JSON.parse(submission.payloadJson) as Record<string, string>,
   755→      createdAt: submission.createdAt,
   756→    };
   757→  }
   758→
   759→  for (const step of PROCEDURE_STEPS[procedureType]) {
   760→    if (stepPayloads[step.stepKey]) continue;
   761→    stepPayloads[step.stepKey] = {
   762→      stepKey: step.stepKey,
   763→      values: {},
   764→      createdAt: null,
   765→    };
   766→  }
   767→
   768→  return { stepPayloads, stepForms };
   769→}

   824→export function submitWorkflowStep(
   825→  requestId: number,
   826→  submittedBy: number,
   827→  payload: Record<string, string>,
   828→): WorkflowRequestDetail | null {
   829→  const db = getDb();
   830→  const request = db
   831→    .prepare(
   832→      `SELECT
   833→        id,
   834→        dept_code as deptCode,
   835→        field_name as fieldName,
   836→        hiring_type as hiringType,
   837→        priority,
   838→        requested_count as requestedCount,
   839→        allocation_result as allocationResult,
   840→        hiring_reason as hiringReason,
   841→        kri_field_name as kriFieldName,
   842→        dept_request_reason as deptRequestReason,
   843→        subject_assignment as subjectAssignment,
   844→        meeting_notes as meetingNotes,
   845→        notes,
   846→        status,
   847→        next_action as nextAction,
   848→        requested_general as requestedGeneral,
   849→        requested_dedicated as requestedDedicated,
   850→        requested_foreign as requestedForeign
   851→      FROM recruitment_requests
   852→      WHERE id = ?`,
   853→    )
   854→    .get(requestId) as RecruitmentRequest | undefined;
   855→
   856→  if (!request) return null;
   857→
   858→  const currentForm = getWorkflowConfigForRequest(request);
   859→  if (!currentForm) {
   860→    throw new Error("현재 단계에 연결된 제출 폼이 없습니다.");
   861→  }
   862→
   863→  for (const field of currentForm.fields) {
   864→    const value = String(payload[field.name] ?? "").trim();
   865→    if (field.required && !value) {
   866→      throw new Error(`${field.label}은(는) 필수 입력 항목입니다.`);
   867→    }
   868→  }
   869→
   870→  const nextStatus = getNextStatus(request.hiringType, request.status);
   871→  const nextAction = nextStatus
   872→    ? getProcedureStepsForHiringType(request.hiringType).find((step) => step.stepKey === nextStatus)?.ownerAction ?? null
   873→    : request.nextAction;
   874→
   875→  const transaction = db.transaction(() => {
   876→    db.prepare(
   877→      `INSERT INTO submission_data (
   878→        request_id,
   879→        submitted_by,
   880→        procedure_type,
   881→        step_key,
   882→        status_before,
   883→        status_after,
   884→        payload_json
   885→      ) VALUES (?, ?, ?, ?, ?, ?, ?)`,
   886→    ).run(
   887→      request.id,
   888→      submittedBy,
   889→      getProcedureType(request.hiringType),
   890→      request.status,
   891→      request.status,
   892→      nextStatus,
   893→      JSON.stringify(payload),
   894→    );
   895→
   896→    if (nextStatus) {
   897→      db.prepare(
   898→        `UPDATE recruitment_requests
   899→         SET status = ?, next_action = ?, updated_at = CURRENT_TIMESTAMP
   900→         WHERE id = ?`,
   901→      ).run(nextStatus, nextAction, request.id);
   902→    }
   903→  });
   904→
   905→  transaction();
   906→  return getRecruitmentRequestDetail(requestId);
   907→}

  1015→    stepKey: string;

  1063→        createdAt: submission?.createdAt ?? null,

  1065→    }
  1066→  }

  1121→        createdAt: submission.createdAt,

  1123→    }
  1124→  }

  1159→        values[`${step.stepKey}__${field.name}`] = submission?.payload[field.name] ?? "";
  1160→      }
  1161→    }
  1162→
  1163→    return {
  1164→      requestId: request.requestId,
  1165→      college: request.college,
  1166→      deptName: request.deptName,
  1167→      fieldName: request.fieldName,
  1168→      hiringType: request.hiringType,
  1169→      currentStatus: request.currentStatus,
  1170→      values,
  1171→      updatedAtByStep,
  1172→    };
  1173→  });

  1176→}

  1181→  payloadJson: string,

  1216→export function updateAdminWorkflowMatrixRow(requestId: number, values: Record<string, string>) {
  1217→  const db = getDb();
  1218→  const request = db
  1219→    .prepare(`SELECT id, hiring_type as hiringType FROM recruitment_requests WHERE id = ?`)
  1220→    .get(requestId) as { id: number; hiringType: string } | undefined;
  1221→  if (!request) return null;
  1222→
  1223→  const procedureType = getProcedureType(request.hiringType);
  1224→  const workflow = WORKFLOW_FORMS[procedureType];
  1225→
  1226→  const groupedByStep = new Map<string, Record<string, string>>();
  1227→  for (const [compoundKey, value] of Object.entries(values)) {
  1228→    const [stepKey, fieldName] = compoundKey.split("__");
  1229→    if (!stepKey || !fieldName || !workflow[stepKey]) continue;
  1230→    const stepPayload = groupedByStep.get(stepKey) ?? {};
  1231→    stepPayload[fieldName] = value;
  1232→    groupedByStep.set(stepKey, stepPayload);
  1233→  }
  1234→
  1235→  const transaction = db.transaction(() => {
  1236→    for (const [stepKey, payload] of groupedByStep.entries()) {
  1237→      upsertAdminWorkflowPayload(requestId, stepKey, JSON.stringify(payload));
  1238→    }
  1239→  });
  1240→  transaction();
  1241→  return true;
  1242→}
