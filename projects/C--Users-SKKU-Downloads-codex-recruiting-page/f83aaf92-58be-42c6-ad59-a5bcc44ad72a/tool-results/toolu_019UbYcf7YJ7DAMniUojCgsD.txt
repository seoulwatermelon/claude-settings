     1→# 성균관대 교원채용 시스템 리팩토링 구현 계획
     2→
     3→> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.
     4→
     5→**Goal:** 한 화면 집중 구조를 딥그린 사이드바 멀티페이지로 전환하고, 밝은 토스증권 스타일 디자인과 DB 기반 권한 매트릭스를 적용한다.
     6→
     7→**Architecture:** Next.js 16 App Router의 Route Groups로 페이지를 분류하고, 공통 layout.tsx에서 사이드바를 렌더링한다. 권한은 SQLite permissions 테이블에서 매 요청마다 조회하며, 기존 52KB dashboard-shell.tsx는 3개 페이지 + 공통 컴포넌트로 분해한다.
     8→
     9→**Tech Stack:** Next.js 16, React 19, TypeScript 5, Tailwind CSS 4, SQLite (better-sqlite3), Pretendard 폰트
    10→
    11→**Spec:** `docs/superpowers/specs/2026-03-18-recruiting-system-refactor-design.md`
    12→
    13→### 스펙 대비 구현 참고사항
    14→
    15→- **DB 컬럼명:** 스펙의 마이그레이션 SQL은 `password_hash`를 사용하지만, 실제 스키마는 `password`이다. 이 계획은 실제 스키마(`password`)를 따른다.
    16→- **checkPermission 동기/비동기:** 스펙은 `async/Promise<boolean>`으로 정의했지만, `better-sqlite3`가 동기 라이브러리이므로 이 계획에서는 동기 함수(`boolean` 반환)로 구현한다.
    17→- **DEAN 사이드바 "단계별 제출":** 스펙은 `/actions/[requestId]`로 매핑하지만, 동적 라우트라 사이드바 정적 링크가 불가능하다. `/progress` 페이지에서 개별 건을 클릭하여 접근하는 방식으로 구현한다.
    18→- **ADMIN의 procedures/actions 권한:** 스펙에 명시되지 않았으나, ADMIN이 채용 절차를 관리하려면 필요하므로 이 계획에서 추가한다.
    19→- **`etc` 리소스:** 스펙에 명시된 "기타" 리소스는 권한 매트릭스 UI의 "기타" 그룹과 시드 데이터에 포함한다.

   420→`app/layout.tsx`를 수정한다. 로그인 페이지는 `(auth)` route group을 사용하므로 이 레이아웃의 영향을 받지 않는다. 단, Next.js App Router에서 root layout은 모든 페이지에 적용되므로, 사이드바 표시 여부를 서버에서 판단한다:
   421→
   422→```tsx
   423→// app/layout.tsx
   424→import type { Metadata } from "next";
   425→import "./globals.css";
   426→import { getCurrentUser } from "@/lib/auth";
   427→import Sidebar from "@/components/sidebar";
   428→
   429→export const metadata: Metadata = {
   430→  title: "SKKU 교원채용 All-In-One",
   431→  description: "성균관대학교 교원채용 정시/상시 통합 관리자 및 보고 페이지",
   432→};
   433→
   434→export default async function RootLayout({
   435→  children,
   436→}: Readonly<{
   437→  children: React.ReactNode;
   438→}>) {
   439→  const user = await getCurrentUser();
   440→
   441→  return (
   442→    <html lang="ko">
   443→      <body>
   444→        {user ? (
   445→          <div style={{ display: "flex", minHeight: "100vh" }}>
   446→            <Sidebar user={user} />
   447→            <main
   448→              style={{
   449→                flex: 1,
   450→                marginLeft: 220,
   451→                padding: "24px 28px",
   452→                minHeight: "100vh",
   453→                background: "var(--page-bg)",
   454→              }}
   455→            >
   456→              {children}
   457→            </main>
   458→          </div>
   459→        ) : (
   460→          children
   461→        )}
   462→      </body>
   463→    </html>
   464→  );
   465→}
   466→```

   704→import { getDb } from "./db";

   706→export function checkPermission(
   707→  role: string,
   708→  resource: string,
   709→  action: string
   710→): boolean {
   711→  const db = getDb();
   712→
   713→  // 먼저 와일드카드 '*' 체크
   714→  const wildcard = db
   715→    .prepare(
   716→      "SELECT allowed FROM permissions WHERE role = ? AND resource = '*' AND action = ? LIMIT 1"
   717→    )
   718→    .get(role, action) as { allowed: number } | undefined;
   719→
   720→  if (wildcard) return wildcard.allowed === 1;
   721→
   722→  // 정확한 resource 체크
   723→  const exact = db
   724→    .prepare(
   725→      "SELECT allowed FROM permissions WHERE role = ? AND resource = ? AND action = ? LIMIT 1"
   726→    )
   727→    .get(role, resource, action) as { allowed: number } | undefined;
   728→
   729→  if (exact) return exact.allowed === 1;
   730→
   731→  // 권한 레코드가 없으면 거부
   732→  return false;
   733→}

   735→export function getPermissionsForRole(role: string): {
   736→  resource: string;
   737→  action: string;
   738→  allowed: boolean;
   739→}[] {
   740→  const db = getDb();
   741→  const rows = db
   742→    .prepare("SELECT resource, action, allowed FROM permissions WHERE role = ?")
   743→    .all(role) as { resource: string; action: string; allowed: number }[];
   744→  return rows.map((r) => ({
   745→    resource: r.resource,
   746→    action: r.action,
   747→    allowed: r.allowed === 1,
   748→  }));
   749→}

   751→export function updatePermissions(
   752→  role: string,
   753→  changes: { resource: string; action: string; allowed: boolean }[]
   754→): void {
   755→  const db = getDb();
   756→  const upsert = db.prepare(`
   757→    INSERT INTO permissions (role, resource, action, allowed)
   758→    VALUES (?, ?, ?, ?)
   759→    ON CONFLICT(role, resource, action)
   760→    DO UPDATE SET allowed = excluded.allowed, updated_at = CURRENT_TIMESTAMP
   761→  `);
   762→  const tx = db.transaction(() => {
   763→    for (const c of changes) {
   764→      upsert.run(role, c.resource, c.action, c.allowed ? 1 : 0);
   765→    }
   766→  });
   767→  tx();
   768→}

   808→  }

   810→  return { success: true };
