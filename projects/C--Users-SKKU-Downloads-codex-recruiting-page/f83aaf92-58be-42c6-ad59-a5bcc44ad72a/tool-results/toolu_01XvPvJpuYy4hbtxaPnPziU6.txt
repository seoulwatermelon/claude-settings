     2→
     3→> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

     5→**Goal:** 한 화면 집중 구조를 딥그린 사이드바 멀티페이지로 전환하고, 밝은 토스증권 스타일 디자인과 DB 기반 권한 매트릭스를 적용한다.

   685→```typescript
   686→// lib/permissions.ts
   687→import { getDb } from "./db";
   688→
   689→export function checkPermission(
   690→  role: string,
   691→  resource: string,
   692→  action: string
   693→): boolean {
   694→  const db = getDb();
   695→
   696→  // 먼저 와일드카드 '*' 체크
   697→  const wildcard = db
   698→    .prepare(
   699→      "SELECT allowed FROM permissions WHERE role = ? AND resource = '*' AND action = ? LIMIT 1"
   700→    )
   701→    .get(role, action) as { allowed: number } | undefined;
   702→
   703→  if (wildcard) return wildcard.allowed === 1;
   704→
   705→  // 정확한 resource 체크
   706→  const exact = db
   707→    .prepare(
   708→      "SELECT allowed FROM permissions WHERE role = ? AND resource = ? AND action = ? LIMIT 1"
   709→    )
   710→    .get(role, resource, action) as { allowed: number } | undefined;
   711→
   712→  if (exact) return exact.allowed === 1;
   713→
   714→  // 권한 레코드가 없으면 거부
   715→  return false;
   716→}
   717→
   718→export function getPermissionsForRole(role: string): {
   719→  resource: string;
   720→  action: string;
   721→  allowed: boolean;
   722→}[] {
   723→  const db = getDb();
   724→  const rows = db
   725→    .prepare("SELECT resource, action, allowed FROM permissions WHERE role = ?")
   726→    .all(role) as { resource: string; action: string; allowed: number }[];
   727→  return rows.map((r) => ({
   728→    resource: r.resource,
   729→    action: r.action,
   730→    allowed: r.allowed === 1,
   731→  }));
   732→}
   733→
   734→export function updatePermissions(
   735→  role: string,
   736→  changes: { resource: string; action: string; allowed: boolean }[]
   737→): void {
   738→  const db = getDb();
   739→  const upsert = db.prepare(`
   740→    INSERT INTO permissions (role, resource, action, allowed)
   741→    VALUES (?, ?, ?, ?)
   742→    ON CONFLICT(role, resource, action)
   743→    DO UPDATE SET allowed = excluded.allowed, updated_at = CURRENT_TIMESTAMP
   744→  `);
   745→  const tx = db.transaction(() => {
   746→    for (const c of changes) {
   747→      upsert.run(role, c.resource, c.action, c.allowed ? 1 : 0);
   748→    }
   749→  });
   750→  tx();
   751→}

   757→    .all() as { role: string }[];
   758→  return rows.map((r) => r.role);
   759→}

   762→  const db = getDb();

   765→    "candidates", "procedures", "actions",

   767→    "system.permissions", "system.users",

   770→  const insert = db.prepare(

   784→  const db = getDb();

   790→  }
   791→  db.prepare("DELETE FROM permissions WHERE role = ?").run(role);
   792→  return { success: true };
   793→}

   866→// app/api/permissions/route.ts
   867→import { NextRequest, NextResponse } from "next/server";
   868→import { requireUser } from "@/lib/auth";
   869→import { checkPermission } from "@/lib/permissions";
   870→import {
   871→  getPermissionsForRole,
   872→  updatePermissions,
   873→  getAllRoles,
   874→} from "@/lib/permissions";
   875→
   876→export async function GET() {
   877→  const user = await requireUser();
   878→  if (!checkPermission(user.role, "system.permissions", "view")) {
   879→    return NextResponse.json({ error: "FORBIDDEN" }, { status: 403 });
   880→  }
   881→
   882→  const roles = getAllRoles();
   883→  const result: Record<string, { resource: string; action: string; allowed: boolean }[]> = {};
   884→  for (const role of roles) {
   885→    result[role] = getPermissionsForRole(role);
   886→  }
   887→  return NextResponse.json({ roles, permissions: result });
   888→}
   889→
   890→export async function PATCH(req: NextRequest) {
   891→  const user = await requireUser();
   892→  if (!checkPermission(user.role, "system.permissions", "edit")) {
   893→    return NextResponse.json({ error: "FORBIDDEN" }, { status: 403 });
   894→  }
   895→
   896→  const body = await req.json();
   897→  const { role, changes } = body as {
   898→    role: string;
   899→    changes: { resource: string; action: string; allowed: boolean }[];
   900→  };
   901→
   902→  updatePermissions(role, changes);
   903→  return NextResponse.json({ success: true });
   904→}

  1051→  function isAllowed(role: string, resource: string, action: string): boolean {
  1052→    const perms = localPerms[role] || [];
  1053→    // 와일드카드 체크
  1054→    const wildcard = perms.find((p) => p.resource === "*" && p.action === action);
  1055→    if (wildcard) return wildcard.allowed;
  1056→    const entry = perms.find((p) => p.resource === resource && p.action === action);
  1057→    return entry?.allowed ?? false;
  1058→  }
  1059→
  1060→  function toggle(resource: string, action: string) {
  1061→    const current = isAllowed(activeRole, resource, action);
  1062→    setLocalPerms((prev) => {
  1063→      const rolePerms = [...(prev[activeRole] || [])];
  1064→      const idx = rolePerms.findIndex(
  1065→        (p) => p.resource === resource && p.action === action
  1066→      );
  1067→      if (idx >= 0) {
  1068→        rolePerms[idx] = { ...rolePerms[idx], allowed: !current };
  1069→      } else {
  1070→        rolePerms.push({ resource, action, allowed: !current });
  1071→      }
  1072→      return { ...prev, [activeRole]: rolePerms };
  1073→    });
  1074→  }

  1081→
  1082→    for (const c of current) {
  1083→      const orig = original.find(
  1084→        (o) => o.resource === c.resource && o.action === c.action
  1085→      );
  1086→      if (!orig || orig.allowed !== c.allowed) {
  1087→        changes.push(c);
  1088→      }
  1089→    }

  1091→    if (changes.length > 0) {

  1093→        method: "PATCH",
  1094→        headers: { "Content-Type": "application/json" },
  1095→        body: JSON.stringify({ role: activeRole, changes }),
